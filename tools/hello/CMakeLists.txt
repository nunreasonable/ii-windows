cmake_minimum_required(VERSION 3.20)
project(iiw_hello LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 6.11 REQUIRED COMPONENTS Gui Quick)
qt_standard_project_setup(REQUIRES 6.11)
qt_add_executable(iiw_hello main.cpp)
qt_add_qml_module(iiw_hello URI Hello VERSION 1.0 QML_FILES Main.qml)
set_target_properties(iiw_hello PROPERTIES WIN32_EXECUTABLE ON)
target_link_libraries(iiw_hello PRIVATE Qt6::Gui Qt6::Quick windowsapp dwmapi)
